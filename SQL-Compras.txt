SELECT 
    p.protocolo_formatado                   AS protocolo_proc_em_tramitacao,
    p.protocolo_formatado_pesquisa          AS protocolo_formatado_pesquisa,
    p.id_protocolo                          AS id_protocolo,
    p.descricao                             AS protocolo_descricao,
    to_char(p.dta_geracao, 'DD-MM-YYYY HH24:MI') AS protocolo_dta_geracao,
    tp.nome                                 AS tipo_procedimento_nome,
    concat(concat(uf.sigla, ' - '),
           uf.nome)                         AS uf_sigla_nome,
    uno.sigla                               AS unidade_sigla,
    pc.id_procedimento                      AS id_procedimento,
    to_char(atv.dth_abertura, 'DD-MM-YYYY HH24:MI') AS dth_abertura, 
    to_clob(dc.conteudo)                    AS conteudo,
    dc.id_documento                         AS id_documento
    ,usu.nome
    ,sr.nome
    -- COALESCE tenta pegar a data da seção. Se for NULL, pega a data de geração do próprio documento.
    ,COALESCE(
        to_char(secdc.dth_atualizacao, 'DD-MM-YYYY HH24:MI'),
        to_char(p_doc.dta_geracao, 'DD-MM-YYYY HH24:MI')
    ) AS dth_documento
FROM
    admsei.protocolo        p, 
    admsei.protocolo        p_doc, -- NOVO: Tabela para buscar a data de documentos sem secao
    admsei.orgao            o,
    admsei.unidade          uno,
    (select * from 
    admsei.uf   
    where id_uf = 7  )        uf,
    admsei.procedimento       pc,
    admsei.tipo_procedimento  tp,
    admsei.documento          doc,
    -- Pega a data da uitima versão do documento 
    (select scdc.id_documento, vsdc.dth_atualizacao
    from 
        admsei.secao_documento scdc,
        admsei.versao_secao_documento vsdc
    where 
            scdc.sin_principal  = 'S'
        AND vsdc.sin_ultima     = 'S'
        AND scdc.id_secao_documento  = vsdc.id_secao_documento
    )   secdc,
    admsei.serie              sr, 
    admsei.documento_conteudo dc,
    ( select * from  
        admsei.atividade  atva
     where 1 = 1
        AND atva.DTH_CONCLUSAO = (
            select max(atvi.DTH_CONCLUSAO) from
                admsei.atividade  atvi
                where 
                    atvi.id_protocolo = atva.id_protocolo
                    AND atvi.ID_TAREFA = atva.ID_TAREFA 
                and rownum < 2
        )
    )     atv,
    admsei.usuario usu
WHERE
        1 = 1
    AND p.sta_protocolo = 'P'
    AND p.sta_estado IN ( '0', '4' )
    AND ( p.sta_nivel_acesso_global <> 2
           OR p.sta_nivel_acesso_local <> 2)
    AND o.id_unidade = uno.id_unidade
    AND uno.id_orgao = o.id_orgao
    AND uno.sigla <> 'TESTE'
    AND uno.sin_ativo = 'S'


    AND pc.id_procedimento = p.id_protocolo
    AND tp.id_tipo_procedimento = pc.id_tipo_procedimento
    
    AND p.id_protocolo = pc.id_procedimento
    AND pc.id_procedimento = doc.id_procedimento
    AND doc.id_documento = dc.id_documento
    
    -- Relacionamentos do Documento
    AND doc.id_documento    = secdc.id_documento(+)
    AND doc.id_documento    = p_doc.id_protocolo(+) -- NOVO: Associa o documento ao seu protocolo raiz para garantir a data
    
    AND pc.id_procedimento = atv.id_protocolo
    AND atv.ID_TAREFA = 5


    AND p.id_usuario_gerador = usu.id_usuario
   
    AND sr.ID_SERIE = doc.ID_SERIE
    AND p.protocolo_formatado like '%.2025.%'
   
    AND p.protocolo_formatado in (select processo 
                                  from admsei.pca_tese_ebrahim);
